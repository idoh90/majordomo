# The front door wants to be found. The app behind it serves nothing a
# crawler can read — every estate lives in localStorage behind JavaScript and
# a sign-in — so opening the origin up indexes exactly one thing: the page
# that asks to be indexed.
User-agent: *
Allow: /

# The origin below is filled in during the build from site.config.ts. Nothing
# to change on a domain move, and nothing to forget.
Sitemap: __SITE_ORIGIN__/sitemap.xml
